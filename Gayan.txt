    function listenToMemberData(id) {
        db.ref('members/' + id).on('value', snapshot => {
            const data = snapshot.val() || {};
            const txs = data.transactions || {};
            let balance = 0;
            let rows = '';

            const txList = Object.values(txs).sort((a,b) => new Date(b.date) - new Date(a.date));

            txList.forEach(t => {
                const amt = parseFloat(t.amount);
                const isInc = t.type === 'INCOME';
                balance += isInc ? amt : -amt;
                
                rows += `<tr class="${isInc ? 'income' : 'expense'}">
                    <td>${t.date}</td>
                    <td>${t.desc}</td>
                    <td>${isInc ? '+' : '-'}${amt.toFixed(2)}</td>
                </tr>`;
            });

            const balDiv = document.getElementById('disp-balance');
            balDiv.innerText = `වත්මන් ශේෂය: රු. ${balance.toFixed(2)}`;
            balDiv.className = `balance-box ${balance >= 0 ? 'balance-pos' : 'balance-neg'}`;
            document.getElementById('member-tx-table').innerHTML = rows;
        });
    }

    function listenToNotices(memId) {
        db.ref('notices').on('value', snapshot => {
            const notices = snapshot.val() || {};
            let html = '';
            const now = Date.now();

            for (let k in notices) {
                const n = notices[k];
                // කල් ඉකුත් වීමේ වේලාව පරීක්ෂා කිරීම

                    continue; // කල් ඉකුත් වී ඇත්නම් සාමාජිකයාට පෙන්වන්නේ නැත
                }

                if (n.target === 'ALL' || n.target === memId) {
                    html += `<div class="notice-box ${n.color}"><span>${n.msg}</span></div>`;
                }
            }
            document.getElementById('announcements-area').innerHTML = html;
        });
    }

    function loadAdminNotices() {
        db.ref('notices').on('value', snapshot => {
            const notices = snapshot.val() || {};
            let html = '';
            const now = Date.now();

                const n = notices[k];

                        <div class="small-text">අහෝසි වන වේලාව: ${expiryStr}</div>
                    </div>

        if (target === 'ALL') {
            db.ref('members').once('value', snapshot => {
                const members = snapshot.val() || {};
                for (let mId in members) {
                    db.ref(`members/${mId}/transactions`).push(txData);
                }
                alert('සියලුම සාමාජිකයින්ට ගනුදෙනුව එකතු කළා!');
            });
        } else {
            db.ref(`members/${target}/transactions`).push(txData);
            alert('ගනුදෙනුව සුරැකිණි!');
        }
    }

    function submitNotice() {
        const target = document.getElementById('notice-target').value;
        const color = document.getElementById('notice-color').value;
        const msg = document.getElementById('notice-msg').value.trim();
        const expiryVal = document.getElementById('notice-expiry').value;

        if (!msg) return alert('පණිවුඩය ඇතුළත් කරන්න');

        const noticeObj = {
            target,
            color,
            msg,
            createdDate: Date.now(),
            expiryTime: expiryVal ? new Date(expiryVal).getTime() : null // Expiry time එකක් ඇතුළත් කළේ නැත්නම් null වේ.
        };

        db.ref('notices').push(noticeObj, () => {
            alert('නිවේදනය පළ කළා!');
            document.getElementById('notice-msg').value = '';
            document.getElementById('notice-expiry').value = '';
        });
    }
